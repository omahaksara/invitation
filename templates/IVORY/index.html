/** * OMAH AKSARA - UCAPAN DATABASE * Final Version - Auto show all history, case-insensitive, JSONP ready * Deploy: Anyone, Execute as: Me */ function doGet(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Database Ucapan Omah Aksara'); const idKlien = e.parameter.c; const callback = e.parameter.callback; if (!idKlien) { return _output({error: 'Parameter?c=idKlien wajib diisi'}, callback); } const data = sheet.getDataRange().getValues(); const hasil = []; // Mulai dari baris 2, skip header for (let i = 1; i < data.length; i++) { const row = data[i]; const idKlienSheet = row[1]? row[1].toString().trim().toLowerCase() : ''; const statusSensor = row[7]? row[7].toString().trim() : 'Aman'; // KOSONG = AUTO AMAN const idKlienQuery = idKlien.toString().trim().toLowerCase(); // FILTER: idKlien harus sama + status harus Aman if (idKlienSheet === idKlienQuery && statusSensor === 'Aman') { hasil.push({ waktu: row[0] || new Date().toISOString(), nama: row[2] || 'Anonim', ucapan: row[3] || '', hadir: row[4] || 'Hadir', pax: row[5] || 1, code: row[6] || '' }); } } // Sort terbaru di atas hasil.sort((a, b) => new Date(b.waktu) - new Date(a.waktu)); return _output(hasil, callback); } catch (err) { return _output({error: err.message}, e.parameter.callback); } } function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Database Ucapan Omah Aksara'); const data = JSON.parse(e.postData.contents); // Validasi minimal if (!data.idKlien ||!data.nama ||!data.ucapan) { return _output({status: 'error', msg: 'Data tidak lengkap'}, null); } // Auto generate code RC + timestamp const kode = 'RC-' + Date.now(); sheet.appendRow([ new Date(), // A: Timestamp data.idKlien.toString().trim().toLowerCase(), // B: ID Klien - auto lowercase + trim data.nama, // C: Nama data.ucapan, // D: Ucapan data.hadir || 'Hadir', // E: Hadir data.pax || 1, // F: Pax kode, // G: Kode 'Aman' // H: Status Sensor - AUTO AMAN ]); return _output({status: 'ok', code: kode}, null); } catch (err) { return _output({status: 'error', msg: err.message}, null); } } // Helper buat JSONP + JSON biasa function _output(data, callback) { const json = JSON.stringify(data); if (callback) { return ContentService.createTextOutput(callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } }
